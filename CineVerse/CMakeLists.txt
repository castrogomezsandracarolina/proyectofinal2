cmake_minimum_required(VERSION 3.20)
project(CineVerse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
add_executable(CineVerse src/main.cpp)
target_include_directories(CineVerse PRIVATE src)
target_link_libraries(CineVerse PRIVATE Threads::Threads)

# La interfaz Qt es un ejecutable independiente; el backend funciona sin Qt.
option(CINEVERSE_CON_QT "Compilar la interfaz grafica con Qt 6 Widgets" OFF)
if(CINEVERSE_CON_QT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_executable(CineVerseQt src/InterfazQt.cpp)
    target_include_directories(CineVerseQt PRIVATE src)
    target_link_libraries(CineVerseQt PRIVATE Qt6::Widgets Threads::Threads)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(CineVerseTests tests/pruebas.cpp)
    target_include_directories(CineVerseTests PRIVATE src)
    target_link_libraries(CineVerseTests PRIVATE Threads::Threads)
    add_test(NAME pruebas_cineverse COMMAND CineVerseTests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/data/wiki_movie_plots_deduped.csv")
        add_executable(CineVerseRealCSVTests tests/pruebas_dataset_real.cpp)
        target_include_directories(CineVerseRealCSVTests PRIVATE src)
        target_link_libraries(CineVerseRealCSVTests PRIVATE Threads::Threads)
        add_test(NAME prueba_csv_real COMMAND CineVerseRealCSVTests "${CMAKE_CURRENT_SOURCE_DIR}/data/wiki_movie_plots_deduped.csv")
    endif()
endif()
